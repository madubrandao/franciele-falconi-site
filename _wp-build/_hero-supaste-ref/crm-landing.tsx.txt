function ContactItem({
  href,
  label,
  children,
}: {
  href?: string;
  label: string;
  children: ReactNode;
}) {
  const content = (
    <>
      <span className={styles.contactIcon}>{children}</span>
      <span className={styles.breakText}>{label}</span>
    </>
  );

  if (!href) return <div className={styles.contactItem}>{content}</div>;

  return (
    <a href={href} className={styles.contactItem} aria-label={label}>
      {content}
    </a>
  );
}
